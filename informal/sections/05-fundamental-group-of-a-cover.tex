\begin{theorem}[id=5.1]{Covers generate the fundamental group}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily, that $x$ lies in every $U_{i}$, and that
$U_{i} \cap U_{j}$ is path connected for all $i$ and $j$.  Then the subgroups
$\im \fgrp{\incl{U_{i}}}{x}$ generate $\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.iSup_range_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{5.1-basepoint}{Proof}{Generation by based loops}
Let $H$ be the subgroup generated by the images in the statement.
For each point $c$, choose an index $i(c)$ with $c\in U_{i(c)}$ and a
path class $\tau_c:x\longrightarrow c$ inside $U_{i(c)}$.  Such a path
exists since $x\in U_{i(c)}$ and $U_{i(c)}=U_{i(c)}\cap U_{i(c)}$ is
path connected.  For a morphism $\omega:c\longrightarrow d$ of
$\fgpd X$ write
\[
  T(\omega)=\tau_c\ptrans\omega\ptrans\tau_d^{-1},
\]
a loop class at $x$.  The morphisms for which $T(\omega)\in H$ form a
subgroupoid $\mathcal H$: the identities give the trivial loop, inverses
give $T(\omega)^{-1}$, and the middle factors $\tau_d^{-1}\ptrans\tau_d$
cancel when two such morphisms are composed.

Suppose that $\omega$ is represented by a path in $U_i$ from $c$ to $d$.
Choose path classes $\sigma_c:x\longrightarrow c$ in
$U_{i(c)}\cap U_i$ and $\sigma_d:x\longrightarrow d$ in
$U_i\cap U_{i(d)}$.  Both intersections contain $x$ and the required
endpoint and are path connected.  Cancellation of inverse path classes
gives
\[
\begin{aligned}
 T(\omega)
   &=(\tau_c\ptrans\sigma_c^{-1})
        \ptrans(\sigma_c\ptrans\omega\ptrans\sigma_d^{-1})\\
   &\qquad\ptrans(\sigma_d\ptrans\tau_d^{-1}).
\end{aligned}
\]
These three factors are loop classes in $U_{i(c)}$, $U_i$ and
$U_{i(d)}$, respectively.  Each therefore lies in $H$, so
$T(\omega)\in H$.  The subgroupoid $\mathcal H$ contains every image
of a cover member; \itemref{4.1} implies $\mathcal H=\fgpd X$.

For a loop class $\omega$ at $x$ we now have $T(\omega)\in H$.
The class $\tau_x$ is itself a loop inside $U_{i(x)}$, so it lies in $H$.
Hence
\[
  \omega=\tau_x^{-1}\ptrans T(\omega)\ptrans\tau_x\in H.
\]
Thus $H=\fgrp{X}{x}$, as required.  For the two-set case, the hypotheses of
\itemref{5.2} give the hypotheses just used for the family $A,B$; its
pairwise intersections are $A$, $B$, $A\cap B$ and $B\cap A$.  This proves
that corollary as well.
\end{proofpassage}

\begin{corollary}[id=5.2]{Two sets generate the fundamental group}
Suppose that \hypCoverTwo, that \hypPathConnected, and that \hypBasepoint.
Then $\im \fgrp{\incl{A}}{x}$ and $\im \fgrp{\incl{B}}{x}$ generate
$\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.range_map_subtypeVal_sup_eq_top}
\end{corollary}
