The proof uses the following standard lemma several times. TauCeti proves a
more general version for pseudometric spaces, but this form will suffice.

\begin{lemma}[id=1.1]{A Lebesgue number}
Let $K\subseteq\R^n$ be compact and $(V_i)_{i\in I}$ a family of subsets of $K$.
Suppose every point of $K$ has some $V_i$ as a neighborhood.
There is a radius $\rho>0$ such that for every $z\in K$, some $V_i$
contains $B_K(z,\rho)$, the ball of radius $\rho$ about $z$ in $K$.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}
In this case, $(V_i)$ is called a \emph{neighborhood cover} of $K$,
and $\rho$ is called a \emph{Lebesgue number} for the cover.

Let $X$ be a \fcnotion{TopologicalSpace}{topological space} and $S\subseteq X$,
with inclusion \fcnotion[Inclusion map]{ContinuousMap.subtypeVal}{$\incl{S}:S\longrightarrow X$}.
For a \fcnotion{Path}{path} $\gamma$ in $X$, write
\fcnotion[Path concatenation]{Path.trans}{$\gamma\ptrans\delta$} for its concatenation
with a composable path $\delta$.
For $r,s\in[0,1]$, let
\fcnotion[Subpath $\gamma_{[r,s]}$]{Path.subpath}{$\subpath{\gamma}{r}{s}(t)=\gamma((1-t)r+ts)$}
be the subpath from $\gamma(r)$ to $\gamma(s)$.
Write \fcnotion[Homotopy class]{Path.Homotopic.Quotient}{$\pcls{\gamma}$} for its homotopy class.

\begin{proposition}[id=1.2]{Subdividing a path}
Consider a path $\gamma:\uI\longrightarrow X$ and a neighborhood cover
$(U_i)_{i\in I}$ of $X$.
Then there is a finite nondecreasing
sequence $0=t_0\le\cdots\le t_n=1$ in $\uI$ such
that each consecutive subpath $\subpath{\gamma}{t_k}{t_{k+1}}$ has image in some $U_i$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

\begin{proofpassage}{1.2-subdivide}{Proof}{Compactness}
\proofstep{1.2-s01-cover}{Pull back the cover}{By continuity, the preimages
$c_i=\gamma^{-1}(U_i)$ form a neighborhood cover of $\uI$.}
\proofstep{1.2-s01-lebesgue}{Obtain a Lebesgue number}{Since $\uI$ is compact, \itemref{1.1} gives a Lebesgue number $\rho>0$.}
\proofstep{1.2-s01-partition}{Choose a finite subdivision with sufficiently short intervals}{Choose a finite subdivision
$0=t_0\le\cdots\le t_N=1$ with each interval of length at most $\rho/2$.}
\proofstep{1.2-s01-segments}{Each interval lies in a cover member}{Each interval $[t_k,t_{k+1}]$ lies in the ball $B_{\uI}(t_k,\rho)$
in $\uI$, which is contained in some $c_i$.}
\proofstep{1.2-s02-subpaths}{Pass from parameter intervals to subpaths}{Hence each
subpath $\subpath{\gamma}{t_k}{t_{k+1}}$ has image in $U_i$.}
\end{proofpassage}

We recover the original path class from the subdivision in \itemref{1.2}
by repeatedly applying the following identity.

\begin{lemma}[id=1.3]{Composing subpath classes}
Let $\gamma$ be a path in $X$, and let $r,s,t$ be parameters.  Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}
