\begin{lemma}[id=1.1]{A Lebesgue number for a family of neighborhoods}
Suppose that $S$ is compact and that each point of $S$ has some $U_{i}$ as a
neighborhood.  Then there is a positive radius $\rho$ such that for each point
$x$ of $S$ some $U_{i}$ contains $\ball{x}{\rho}$.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}

\begin{lemma}[id=1.2]{Equally spaced subdivision points of an interval}
Let $d$ be nonnegative and let $n$ be a natural number.  Then the point reached
from $a$ by $n$ steps of size $d$ is the smaller of $b$ and $a + nd$.
\leanref{Set.Icc.coe_addNSMul}
\end{lemma}

\begin{definition}[id=1.3]{The restriction of a path to a subspace}
Let $x$ and $y$ be points of $S$, and let $\gamma$ be a path in $X$ from $x$ to
$y$ every value of which lies in $S$.  Then $\codres{\gamma}{S}$ is the path in
$S$ from $x$ to $y$ whose value at a parameter $t$ is $\gamma(t)$.
\leanref{Path.codRestrict}
\leanref{Path.codRestrict_coe}
\end{definition}

\begin{lemma}[id=1.4]{Mapping a restricted path back to the ambient space}
Let $x$ and $y$ be points of $S$, and let $\gamma$ be a path in $X$ from $x$ to
$y$ every value of which lies in $S$.  Then $\incl{S} \circ \codres{\gamma}{S} =
\gamma$.
\leanref{Path.map_codRestrict}
\end{lemma}

\begin{lemma}[id=1.5]{The values of a concatenation of paths}
Let $\gamma$ be a path from $x$ to $y$, let $\delta$ be a path from $y$ to $z$,
and let $u$ and $v$ be parameters.
\itempart{a}{%
  If $u \le 1/2$ and $v = 2u$, then
  $(\gamma \ptrans \delta)(u) = \gamma(v)$.
  \leanref{Path.trans_apply_of_le}}
\itempart{b}{%
  If $1/2 \le u$ and $v = 2u - 1$, then
  $(\gamma \ptrans \delta)(u) = \delta(v)$.
  \leanref{Path.trans_apply_of_ge}}
\end{lemma}

\begin{lemma}[id=1.6]{A subpath of a path confined to a set}
Let $\gamma$ be a path in $X$, let $r$ and $s$ be parameters, and suppose that
$\gamma(t)$ lies in $S$ for every parameter $t$ with $r \le t \le s$.  Then, for
all parameters $u$ and $v$ with $r \le u \le s$ and $r \le v \le s$, every value
of $\subpath{\gamma}{u}{v}$ lies in $S$.
\leanref{Path.subpath_apply_mem}
\end{lemma}

\begin{proposition}[id=1.7]{Subdividing a path along a family of sets}
Let $\gamma$ be a path in $X$, and suppose that every parameter has a
neighborhood that $\gamma$ maps into some $U_{i}$.  Then there are a natural
number $n$ and a subdivision $t_{0}, \dots, t_{n}$ of $\uI$ with $t_{0} = 0$ and
$t_{n} = 1$ such that, for each $k$ less than $n$, the range of
$\subpath{\gamma}{t_{k}}{t_{k+1}}$ is contained in some $U_{i}$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

\begin{lemma}[id=1.8]{Cutting a path class at an intermediate parameter}
Let $\gamma$ be a path in $X$, let $r$, $s$ and $t$ be parameters, and let
$x_{0}$, $x_{1}$ and $x_{2}$ be points of $X$ with $x_{0} = \gamma(r)$,
$x_{1} = \gamma(s)$ and $x_{2} = \gamma(t)$.  Read $\subpath{\gamma}{r}{s}$ as a
path from $x_{0}$ to $x_{1}$, $\subpath{\gamma}{s}{t}$ as a path from $x_{1}$ to
$x_{2}$, and $\subpath{\gamma}{r}{t}$ as a path from $x_{0}$ to $x_{2}$, by
renaming their endpoints.  Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}
