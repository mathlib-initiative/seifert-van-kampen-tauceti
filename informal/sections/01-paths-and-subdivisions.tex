Let $X$ be a \fcnotion{TopologicalSpace}{topological space} and $S\subseteq X$,
with inclusion \fcnotion{ContinuousMap.subtypeVal}{$\incl{S}:S\longrightarrow X$}.
For a \fcnotion{Path}{path} $\gamma$ in $X$, write
\fcnotion{Path.trans}{$\gamma\ptrans\delta$} for its concatenation
with a composable path $\delta$.
For $r,s\in[0,1]$, the subpath
\fcnotion{Path.subpath}{$\subpath{\gamma}{r}{s}$} traverses $\gamma$ with
its parameter running from $r$ to $s$:
$\subpath{\gamma}{r}{s}(t)=\gamma((1-t)r+ts)$ for $t\in[0,1]$.
Write \fcnotion{Path.Homotopic.Quotient}{$\pcls{\gamma}$} for its homotopy
class, with \fcnotion{Path.Homotopic.Quotient.trans}{composition} again
denoted $\ptrans$.

\begin{proposition}[id=1.1]{Subdividing a path}
Let $\gamma$ be a path in $X$ and $(U_i)_{i\in I}$ a family of subsets of $X$.
Suppose that every parameter has a neighborhood that $\gamma$ maps into
some $U_i$.  Then there is a finite \fcnotion{Monotone}{nondecreasing}
sequence $0=t_0\le\cdots\le t_n=1$ in \fcnotion{unitInterval}{$\uI$} such
that each consecutive subpath $\subpath{\gamma}{t_k}{t_{k+1}}$ has image in some $U_i$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

\begin{proofpassage}{1.1-subdivide}{Proof}{Compactness}
\proofstep{1.1-s01-cover}{The preimage interiors form an open cover}{Put
$V_i=\operatorname{int}_{\uI}(\gamma^{-1}(U_i))$; the hypothesis says that
the $V_i$ form an open cover of $\uI$.}
Since $\uI$ is compact, \itemref{1.3} gives a Lebesgue radius $\rho>0$.
Choose a positive integer $N$ with $1/N<\rho$ and set $t_k=k/N$.
Each interval $[t_k,t_{k+1}]$ lies in $\uI\cap\ball{t_k}{\rho}$,
which is contained in some $V_i$.
\proofstep{1.1-s02-subpaths}{Pass from parameter intervals to subpaths}{Hence each
subpath $\subpath{\gamma}{t_k}{t_{k+1}}$ has image in $U_i$.}
\end{proofpassage}

\begin{lemma}[id=1.2]{Composing subpath classes}
Let $\gamma$ be a path in $X$, and let $r,s,t$ be parameters.  Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}

\begin{lemma}[id=1.3]{A Lebesgue number}
Let $K\subseteq\R^n$ be \fcnotion{IsCompact}{compact}, and let
$(V_j)_{j\in J}$ be a family of subsets of $K$.  If each point of $K$ has
some $V_j$ as a neighborhood in $K$, then there is a radius $\rho>0$ such
that for each $z\in K$ some $V_j$ contains
\fcnotion{Metric.ball}{$K\cap\ball{z}{\rho}$}.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}
