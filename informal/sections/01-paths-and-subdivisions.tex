\begin{lemma}[id=1.1]{A Lebesgue number for a family of neighborhoods}
Let $X$ be a pseudometric space, let $I$ be an index type, let $S$ be a subset
of $X$, and let $U$ be a family of subsets of $X$ indexed by $I$.  Suppose that
$S$ is compact, and that every point $x$ of $S$ has some $U_{i}$ as a
neighborhood.  Then there is a real number $\rho$ greater than zero with the
following property: for every point $x$ of $S$ there is an index $i$ such that
$\ball{x}{\rho}$ is contained in $U_{i}$.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}

\begin{lemma}[id=1.2]{Equally spaced subdivision points of an interval}
Let $G$ be an additive commutative group that is linearly ordered and whose
addition respects that order.  Let $a$, $b$ and $d$ be elements of $G$ with $a$
at most $b$ and $d$ at least $0$, and let $n$ be a natural number.  The element
of $G$ underlying the point reached from $a$ by $n$ steps of size $d$ is the
smaller of $b$ and $a + nd$.
\leanref{Set.Icc.coe_addNSMul}
\end{lemma}

\begin{definition}[id=1.3]{The restriction of a path to a subspace}
Let $S$ be a subset of $X$, let $x$ and $y$ be points of $S$, and let $\gamma$
be a path in $X$ from the point of $X$ under $x$ to the point of $X$ under $y$.
Suppose that $\gamma(t)$ lies in $S$ for every parameter $t$.  Then
$\codres{\gamma}{S}$ is the path in $S$ from $x$ to $y$ whose value at a
parameter $t$ is the point of $S$ carried by $\gamma(t)$.
\leanref{Path.codRestrict}
The point of $X$ under $\codres{\gamma}{S}(t)$ is $\gamma(t)$, for every
parameter $t$.
\leanref{Path.codRestrict_coe}
\end{definition}

\begin{lemma}[id=1.4]{Mapping a restricted path back to the ambient space}
Let $S$ be a subset of $X$, let $x$ and $y$ be points of $S$, and let $\gamma$
be a path in $X$ from the point of $X$ under $x$ to the point of $X$ under $y$
such that $\gamma(t)$ lies in $S$ for every parameter $t$.  The image of
$\codres{\gamma}{S}$ under $\incl{S}$ is $\gamma$.
\leanref{Path.map_codRestrict}
\end{lemma}

\begin{lemma}[id=1.5]{The values of a concatenation of paths}
Let $x$, $y$ and $z$ be points of $X$, let $\gamma$ be a path from $x$ to $y$,
and let $\delta$ be a path from $y$ to $z$.
\itempart{a}{%
  Let $u$ be a parameter whose value in $\R$ is at most $1/2$.  For every
  parameter $v$ whose value in $\R$ is $2u$, the value of $\gamma \ptrans
  \delta$ at $u$ is $\gamma(v)$.
  \leanref{Path.trans_apply_of_le}}
\itempart{b}{%
  Let $u$ be a parameter whose value in $\R$ is at least $1/2$.  For every
  parameter $v$ whose value in $\R$ is $2u - 1$, the value of $\gamma \ptrans
  \delta$ at $u$ is $\delta(v)$.
  \leanref{Path.trans_apply_of_ge}}
\end{lemma}

\begin{lemma}[id=1.6]{A subpath of a path confined to a set}
Let $x$ and $y$ be points of $X$, let $\gamma$ be a path from $x$ to $y$, and
let $S$ be a subset of $X$.  Let $r$ and $s$ be parameters, and suppose that
$\gamma(t)$ lies in $S$ for every parameter $t$ with $r \le t \le s$.  Let $u$
and $v$ be parameters with $r \le u \le s$ and $r \le v \le s$.  Then
$\subpath{\gamma}{u}{v}(t)$ lies in $S$ for every parameter $t$.
\leanref{Path.subpath_apply_mem}
\end{lemma}

\begin{proposition}[id=1.7]{Subdividing a path along a family of sets}
Let $I$ be an index type, let $U$ be a family of subsets of $X$ indexed by $I$,
let $x$ and $y$ be points of $X$, and let $\gamma$ be a path from $x$ to $y$.
Suppose that for every parameter $s$ there is an index $i$ such that the
preimage of $U_{i}$ under $\gamma$ is a neighborhood of $s$.  Then there are a
natural number $n$ and parameters $t_{0}, \dots, t_{n}$ such that $t_{0} = 0$,
$t_{n} = 1$ and $t_{0} \le t_{1} \le \dots \le t_{n}$, and such that for every
natural number $k$ smaller than $n$ there is an index $i$ for which the range
of $\subpath{\gamma}{t_{k}}{t_{k+1}}$ is contained in $U_{i}$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

\begin{lemma}[id=1.8]{Cutting a path class at an intermediate parameter}
Let $x$ and $y$ be points of $X$, and let $\gamma$ be a path from $x$ to $y$.
Let $r$, $s$ and $t$ be parameters, and let $x_{0}$, $x_{1}$ and $x_{2}$ be
points of $X$ with $x_{0} = \gamma(r)$, $x_{1} = \gamma(s)$ and $x_{2} =
\gamma(t)$.  Read $\subpath{\gamma}{r}{s}$ as a path from $x_{0}$ to $x_{1}$,
read $\subpath{\gamma}{s}{t}$ as a path from $x_{1}$ to $x_{2}$, and read
$\subpath{\gamma}{r}{t}$ as a path from $x_{0}$ to $x_{2}$, each along these
three equations.  Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}
