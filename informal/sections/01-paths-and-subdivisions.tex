We begin with a \fcnotion{IsCompact}{compactness} lemma for
\fcnotion{PseudoMetricSpace}{pseudometric spaces};
\fcnotion{Metric.ball}{$\ball{z}{\rho}$} denotes an open ball.

\begin{lemma}[id=1.1]{A Lebesgue number for a family of neighborhoods}
Let $Z$ be a pseudometric space, $K$ a compact subset of $Z$, and
$(V_j)_{j\in J}$ a family of subsets of $Z$.  If each point of $K$ has some
$V_j$ as a neighborhood, then there is a positive radius $\rho$ such that for
each $z\in K$ some $V_j$ contains $\ball{z}{\rho}$.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}

For the subdivision formula, let $G$ be an
\fcnotion{AddCommGroup}{additive commutative group} with a
\fcnotion{LinearOrder}{linear order}
\fcnotion{IsOrderedAddMonoid}{preserved by addition}, and let $a\le b$ be
elements of $G$.  Write \fcnotion{Set.Icc}{$\cint{a}{b}$} for the closed
interval from $a$ to $b$.  For $d\in G$ and a \fcnotion{Nat}{nonnegative integer} $n$,
the \fcnotion{Set.Icc.addNSMul}{subdivision point} reached from $a$ by $n$ steps of size $d$
is $a+nd$, replaced by $a$ if it is below $a$ and by $b$ if it is above $b$;
$nd$ is the sum of $n$ copies of $d$.

\begin{lemma}[id=1.2]{Equally spaced subdivision points of an interval}
Let $d$ be nonnegative.  Then the point reached from $a$ by $n$ steps of size
$d$ is the smaller of $b$ and $a + nd$.
\leanref{Set.Icc.coe_addNSMul}
\end{lemma}

Let $S$ be a subset of a \fcnotion{TopologicalSpace}{topological space} $X$.
Write \fcnotion{ContinuousMap.subtypeVal}{$\incl{S}$} for the inclusion
$S\longrightarrow X$.  A \fcnotion{Path}{path} from $x$ to $y$ in $X$ is a
\fcnotion{ContinuousMap}{continuous map} from the unit interval
\fcnotion{unitInterval}{$\uI$} of \fcnotion{Real}{real numbers} to $X$, with values
$x$ at $0$ and $y$ at $1$.  Path parameters lie in $\uI$.

\begin{definition}[id=1.3]{The restriction of a path to a subspace}
Let $x$ and $y$ be points of $S$, and let $\gamma$ be a path in $X$ from $x$ to
$y$ every value of which lies in $S$.  Then $\codres{\gamma}{S}$ is the path in
$S$ from $x$ to $y$ whose value at a parameter $t$ is $\gamma(t)$.
\leanref{Path.codRestrict}
\leanref{Path.codRestrict_coe}
\end{definition}

\begin{lemma}[id=1.4]{Mapping a restricted path back to the ambient space}
Let $x$ and $y$ be points of $S$, and let $\gamma$ be a path in $X$ from $x$ to
$y$ every value of which lies in $S$.  Then $\incl{S} \circ \codres{\gamma}{S} =
\gamma$.
\leanref{Path.map_codRestrict}
\end{lemma}

For composable paths $\gamma$ and $\delta$, write
\fcnotion{Path.trans}{$\gamma\ptrans\delta$} for their concatenation, with
$\gamma$ traversed first.

\begin{lemma}[id=1.5]{The values of a concatenation of paths}
Let $\gamma$ be a path from $x$ to $y$ and $\delta$ a path from $y$ to $z$ in
$X$, and let $u$ and $v$ be parameters.
\itempart{a}{%
  If $u \le 1/2$ and $v = 2u$, then
  $(\gamma \ptrans \delta)(u) = \gamma(v)$.
  \leanref{Path.trans_apply_of_le}}
\itempart{b}{%
  If $1/2 \le u$ and $v = 2u - 1$, then
  $(\gamma \ptrans \delta)(u) = \delta(v)$.
  \leanref{Path.trans_apply_of_ge}}
\end{lemma}

For a path $\gamma$ and parameters $r,s$, the subpath
\fcnotion{Path.subpath}{$\subpath{\gamma}{r}{s}$} traverses $\gamma$ with its
parameter running linearly from $r$ to $s$.  It runs backwards when $s<r$
and is constant when $r=s$.

\begin{lemma}[id=1.6]{A subpath of a path confined to a set}
Let $\gamma$ be a path in $X$, and let $r$ and $s$ be parameters.
Suppose that $\gamma(t)$ lies in $S$ for every parameter $t$ with
$r \le t \le s$.  Then, for all parameters $u$ and $v$ with $r \le u \le s$ and
$r \le v \le s$, every value of $\subpath{\gamma}{u}{v}$ lies in $S$.
\leanref{Path.subpath_apply_mem}
\end{lemma}

A subdivision of $\uI$ is a finite \fcnotion{Monotone}{nondecreasing} sequence
$0=t_0\le\cdots\le t_n=1$.

Let $(U_i)_{i\in I}$ be a family of subsets of $X$.

\begin{proposition}[id=1.7]{Subdividing a path along a family of sets}
Let $\gamma$ be a path in $X$.  Suppose that every parameter has a
neighborhood that $\gamma$ maps into some $U_i$.  Then there is a subdivision
$t_0,\dots,t_n$ of $\uI$ such that, for each $k$ less than $n$, the range of
$\subpath{\gamma}{t_k}{t_{k+1}}$ is contained in some $U_i$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

Two paths with the same endpoints are homotopic if a homotopy joins them while
fixing the endpoints.  Write \fcnotion{Path.Homotopic.Quotient}{$\pcls{\gamma}$}
for the class of $\gamma$; \fcnotion{Path.Homotopic.Quotient.trans}{composition
of classes}, also written $\ptrans$, is induced by concatenation.  Equalities of endpoints allow a path
to be \fcnotion{Path.cast}{read with the renamed endpoints}.

\begin{lemma}[id=1.8]{Cutting a path class at an intermediate parameter}
Let $\gamma$ be a path in $X$, and let $r,s,t$ be parameters.
Let $x_0$, $x_1$ and $x_2$ be points of $X$ with $x_0 = \gamma(r)$,
$x_1 = \gamma(s)$ and $x_2 = \gamma(t)$.  Rename the endpoints of
$\subpath{\gamma}{r}{s}$, $\subpath{\gamma}{s}{t}$ and $\subpath{\gamma}{r}{t}$
along these equalities.  Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}
