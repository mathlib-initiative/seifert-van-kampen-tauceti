We begin with a \fcnotion{IsCompact}{compactness} lemma for
\fcnotion{PseudoMetricSpace}{pseudometric spaces};
\fcnotion{Metric.ball}{$\ball{z}{\rho}$} denotes an open ball.

\begin{lemma}[id=1.1]{A Lebesgue number for a family of neighborhoods}
Let $Z$ be a pseudometric space, $K$ a compact subset of $Z$, and
$(V_j)_{j\in J}$ a family of subsets of $Z$.  If each point of $K$ has some
$V_j$ as a neighborhood, then there is a positive radius $\rho$ such that for
each $z\in K$ some $V_j$ contains $\ball{z}{\rho}$.
\leanref{TauCeti.lebesgue_number_lemma_of_metric_of_mem_nhds}
\end{lemma}

Let $S$ be a subset of a \fcnotion{TopologicalSpace}{topological space} $X$.
Write \fcnotion{ContinuousMap.subtypeVal}{$\incl{S}$} for the inclusion
$S\longrightarrow X$.

Write \fcnotion{Path.codRestrict}{$\codres{\gamma}{S}$} for $\gamma$ viewed as a
path in $S$ when its image lies there, and
\fcnotion{Path.trans}{$\gamma\ptrans\delta$} for the concatenation of
composable \fcnotion{Path}{paths} $\gamma$ and $\delta$, with $\gamma$ first.
The subpath \fcnotion{Path.subpath}{$\subpath{\gamma}{r}{s}$} traverses
$\gamma$ with its parameter running linearly from $r$ to $s$, backwards when
$s<r$ and constantly when $r=s$.

A subdivision of \fcnotion{unitInterval}{$\uI$} is a finite \fcnotion{Monotone}{nondecreasing} sequence
$0=t_0\le\cdots\le t_n=1$.

Let $(U_i)_{i\in I}$ be a family of subsets of $X$.

\begin{proposition}[id=1.2]{Subdividing a path along a family of sets}
Let $\gamma$ be a path in $X$.  Suppose that every parameter has a
neighborhood that $\gamma$ maps into some $U_i$.  Then there is a subdivision
$t_0,\dots,t_n$ of $\uI$ such that, for each $k$ less than $n$, the range of
$\subpath{\gamma}{t_k}{t_{k+1}}$ is contained in some $U_i$.
\leanref{Path.exists_monotone_range_subpath_subset}
\end{proposition}

Write \fcnotion{Path.Homotopic.Quotient}{$\pcls{\gamma}$} for the homotopy
class of $\gamma$; \fcnotion{Path.Homotopic.Quotient.trans}{composition of
classes}, also written $\ptrans$, is induced by concatenation.

\begin{lemma}[id=1.3]{Cutting a path class at an intermediate parameter}
Let $\gamma$ be a path in $X$, and let $r,s,t$ be parameters.
Then
\[
  \pcls{\subpath{\gamma}{r}{s}} \ptrans \pcls{\subpath{\gamma}{s}{t}}
    = \pcls{\subpath{\gamma}{r}{t}}.
\]
\leanref{Path.Homotopic.Quotient.subpath_cast_trans}
\end{lemma}
