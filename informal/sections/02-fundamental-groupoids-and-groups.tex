\begin{lemma}[id=2.1]{The fundamental groupoid of a path-connected space}
Suppose that $X$ is path connected, and let $x$ and $y$ be objects of
$\fgpd{X}$.  Then at least one morphism of $\fgpd{X}$ runs from $x$ to $y$.
\leanref{FundamentalGroupoid.nonempty_hom}
\end{lemma}

\begin{lemma}[id=2.2]{An injective map is injective on objects}
Let $f$ be a continuous map from $X$ to $Y$, and suppose that $f$ is
injective.  Then $\fgpd{f}$ is injective on objects, that is, two distinct
objects of $\fgpd{X}$ have two distinct images under $\fgpd{f}$.
\leanref{ContinuousMap.fundamentalGroupoid_map_obj_injective}
\end{lemma}

\begin{lemma}[id=2.3]{Recasting the endpoints of a path class}
Let $\mathcal{H}$ be a subgroupoid of $\fgpd{X}$, let $x$, $y$, $x'$ and $y'$
be points of $X$, let $\omega$ be a class of paths from $x$ to $y$, and
suppose that $x' = x$ and $y' = y$.  Along these two equations, $\omega$ is
read as a class of paths from $x'$ to $y'$.  That class lies in the set
$\mathcal{H}$ picks out between $x'$ and $y'$ if and only if $\omega$ lies in
the set $\mathcal{H}$ picks out between $x$ and $y$.
\leanref{Path.Homotopic.Quotient.cast_mem_arrows_iff}
\end{lemma}

\begin{lemma}[id=2.4]{The image of the class of a loop}
Let $f$ be a continuous map from $X$ to $Y$, let $x$ be a point of $X$, and
let $\gamma$ be a loop at $x$.  Write $f \circ \gamma$ for the loop at $f(x)$
whose value at a parameter $t$ is $f(\gamma(t))$.  Then $\fgrp{f}{x}$ carries
the element $\pcls{\gamma}$ of $\fgrp{X}{x}$ to the element
$\pcls{f \circ \gamma}$ of $\fgrp{Y}{f(x)}$.
\leanref{TauCeti.FundamentalGroup.map_fromPath}
\end{lemma}
