\begin{definition}[id=6.1]{The homomorphism out of the fundamental group of the space}
Let $x$ be a point of $X$ and let $A$ and $B$ be subsets of $X$.  Let $M$ be a
monoid.  Suppose that \hypCoverTwo and that \hypPathConnected.
Suppose also that \hypBasepoint.  The point $x$ then lies in $A \cap B$, and it
is the basepoint of $A$, of $B$ and of $A \cap B$.  Let $\varphi_{A}$ be a
homomorphism from $\fgrp{A}{x}$ to $M$, let $\varphi_{B}$ be a homomorphism
from $\fgrp{B}{x}$ to $M$, and suppose that
\[
  \varphi_{A} \circ \fgrp{\inclin{A \cap B}{A}}{x}
    = \varphi_{B} \circ \fgrp{\inclin{A \cap B}{B}}{x} .
\]
These data determine a homomorphism $\varphi$ from $\fgrp{X}{x}$ to $M$, which
we call the homomorphism induced by $\varphi_{A}$ and $\varphi_{B}$.
\leanref{TauCeti.vanKampenDesc}
\end{definition}

\begin{proposition}[id=6.2]{The induced homomorphism restricts to the given ones}
Let $x$ be a point of $X$ and let $A$ and $B$ be subsets of $X$.  Let $M$ be a
monoid.  Suppose that \hypCoverTwo and that \hypPathConnected.
Suppose also that \hypBasepoint.  The point $x$ then lies in $A \cap B$, and it
is the basepoint of $A$, of $B$ and of $A \cap B$.  Let $\varphi_{A}$ be a
homomorphism from $\fgrp{A}{x}$ to $M$, let $\varphi_{B}$ be a homomorphism
from $\fgrp{B}{x}$ to $M$, suppose that
\[
  \varphi_{A} \circ \fgrp{\inclin{A \cap B}{A}}{x}
    = \varphi_{B} \circ \fgrp{\inclin{A \cap B}{B}}{x} ,
\]
and let $\varphi$ be the homomorphism induced by $\varphi_{A}$ and
$\varphi_{B}$.
\itempart{a}{%
  The value of $\varphi$ at the image under $\fgrp{\incl{A}}{x}$ of any
  element $\omega$ of $\fgrp{A}{x}$ is $\varphi_{A}(\omega)$, and
  $\varphi \circ \fgrp{\incl{A}}{x} = \varphi_{A}$.
  \leanref{TauCeti.vanKampenDesc_map_left}%
  \leanref{TauCeti.vanKampenDesc_comp_map_left}}
\itempart{b}{%
  The value of $\varphi$ at the image under $\fgrp{\incl{B}}{x}$ of any
  element $\omega$ of $\fgrp{B}{x}$ is $\varphi_{B}(\omega)$, and
  $\varphi \circ \fgrp{\incl{B}}{x} = \varphi_{B}$.
  \leanref{TauCeti.vanKampenDesc_map_right}%
  \leanref{TauCeti.vanKampenDesc_comp_map_right}}
\end{proposition}

\begin{proposition}[id=6.3]{Uniqueness of the induced homomorphism}
Let $x$ be a point of $X$ and let $A$ and $B$ be subsets of $X$.  Let $M$ be a
monoid.  Suppose that \hypCoverTwo and that \hypPathConnected.
Suppose also that \hypBasepoint.  The point $x$ is then the basepoint of $A$
and of $B$.  Let $\varphi$ and $\psi$ be any two homomorphisms from
$\fgrp{X}{x}$ to $M$.  Neither is assumed to come from a pair of
homomorphisms on $A$ and on $B$.  If
$\varphi \circ \fgrp{\incl{A}}{x} = \psi \circ \fgrp{\incl{A}}{x}$ and
$\varphi \circ \fgrp{\incl{B}}{x} = \psi \circ \fgrp{\incl{B}}{x}$, then
$\varphi = \psi$.
\leanref{TauCeti.vanKampen_hom_ext}
\end{proposition}

\begin{theorem}[id=6.4]{The based Seifert--van Kampen theorem}
\fcref{Theorem 1.20}
Let $x$ be a point of $X$ and let $A$ and $B$ be subsets of $X$.  Suppose that
\hypCoverTwo and that \hypPathConnected.  Suppose also that
\hypBasepoint.  Neither $A$ nor $B$ is assumed to be open.  The point $x$ then
lies in $A \cap B$, and it is the basepoint of $A$, of $B$ and of
$A \cap B$.  The homomorphism
$\fgrp{\inclin{A \cap B}{A}}{x}$ runs from
$\fgrp{A \cap B}{x}$ to $\fgrp{A}{x}$ and the homomorphism
$\fgrp{\inclin{A \cap B}{B}}{x}$ runs from $\fgrp{A \cap B}{x}$ to
$\fgrp{B}{x}$, while the homomorphism $\fgrp{\incl{A}}{x}$ runs from
$\fgrp{A}{x}$ to $\fgrp{X}{x}$ and the homomorphism $\fgrp{\incl{B}}{x}$ runs
from $\fgrp{B}{x}$ to $\fgrp{X}{x}$.  These four homomorphisms form a pushout
square of groups.
\leanref{TauCeti.isPushout_fundamentalGroup}
\end{theorem}
