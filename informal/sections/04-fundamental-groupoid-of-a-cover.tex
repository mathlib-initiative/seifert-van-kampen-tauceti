Let $(U_i)_{i\in I}$ be a family of subsets of a topological space $X$.
Write $\incl{S}:S\longrightarrow X$ for inclusion and $\fgpd{f}$ for the
functor on fundamental groupoids induced by a continuous map $f$.
The image \fcnotion{CategoryTheory.Subgroupoid.im}{$\im\fgpd{\incl{U_i}}$}
consists of the path classes coming from $U_i$.  These image subgroupoids
generate $\fgpd{X}$ when
\fcnotion{iSup}{the smallest subgroupoid containing them} is $\fgpd{X}$.

\begin{theorem}[id=4.1]{Covers generate the fundamental groupoid}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily.  Then the subgroupoids $\im \fgpd{\incl{U_{i}}}$
generate $\fgpd{X}$.
\leanref{TauCeti.FundamentalGroupoid.iSup_im_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{4.1-subdivide}{Proof}{Subdivide and reassemble}
Let $\mathcal H$ be the subgroupoid generated by the images in the
statement, and let $\omega$ be a morphism of $\fgpd X$.  Choose a path
$\gamma$ representing $\omega$.  \proofstep{4.1-s01-subdivide}{Choose a subdivision subordinate to the cover}{Pulling the covering neighborhoods back
along $\gamma$ gives the hypothesis of \itemref{1.3}; hence choose
$0=t_0\le\cdots\le t_n=1$ and indices $i_k$ such that
$\gamma_k=\subpath{\gamma}{t_k}{t_{k+1}}$ lies in $U_{i_k}$.}
\proofstep{4.1-s02-image}{Each piece lies in an image subgroupoid}{Mapping $\codres{\gamma_k}{U_{i_k}}$ back to $X$ recovers
$\gamma_k$.  Thus $[\gamma_k]$ lies in the corresponding image subgroupoid,
and hence in $\mathcal H$.}  Repeatedly applying \itemref{1.4} gives
\proofequation{4.1-s03-reassemble}{Reassemble the original path class}{  \omega=[\gamma]=[\gamma_0]\ptrans\cdots\ptrans[\gamma_{n-1}].}
The right side lies in $\mathcal H$ by closure under composition.  Every
morphism therefore lies in $\mathcal H$, which proves the claim.
\end{proofpassage}

Let $\mathcal D$ be a category.  For subsets $S\subseteq T$ of $X$, write
\fcnotion{ContinuousMap.inclusion}{$\inclin{S}{T}$} for the inclusion
$S\longrightarrow T$.  Functor composition
\fcnotion{CategoryTheory.Functor.comp}{$F\circ G$} is read from right to left.

\begin{definition}[id=4.2]{Gluing functors out of fundamental groupoids}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily.  For each $i$ let $F_{i}$ be a functor from
$\fgpd{U_{i}}$ to $\mathcal{D}$, and suppose that for all $i$ and $j$
\[
  F_{i} \circ \fgpd{\inclin{U_{i} \cap U_{j}}{U_{i}}}
    = F_{j} \circ \fgpd{\inclin{U_{i} \cap U_{j}}{U_{j}}}.
\]
Then the $F_{i}$ glue to a functor $F$ from $\fgpd{X}$ to $\mathcal{D}$.
\leanref{TauCeti.FundamentalGroupoid.glue}
\end{definition}

\begin{proofpassage}{4.2-homotopy}{Construction}{Gluing on homotopy classes}
For a point $y$, choose a cover member containing it and put
$F(y)=F_i(y)$.  \proofstep{4.2-s01-objects}{The object value is independent of the cover member}{This does not depend on $i$: if $y\in U_i\cap U_j$,
the assumed equality of functors on the intersection gives
$F_i(y)=F_j(y)$.}

For a path $\gamma$ from $y$ to $z$, \proofstep{4.2-s02-partition}{Choose a sufficiently fine equal subdivision}{apply \itemref{1.1} to the neighborhoods
$\gamma^{-1}(U_i)$ in the compact interval $\uI$.  For every sufficiently
large positive integer $N$, the equal subdivision $t_k=k/N$ has each piece
$\gamma_k=\subpath{\gamma}{t_k}{t_{k+1}}$ lying in some $U_{i_k}$.}
Call such a subdivision fine, and put
\[
  L_N(\gamma)=F_{i_{N-1}}([\gamma_{N-1}])\circ\cdots
                \circ F_{i_0}([\gamma_0]).
\]
The object equalities above make the factors composable.
\proofstep{4.2-s03-indices}{Compatibility makes the local path value independent of the index}{If a piece lies in both $U_i$ and $U_j$, regard it as a path in their
intersection; compatibility gives the same value under $F_i$ and $F_j$.}
\proofstep{4.2-s04-refine-pieces}{Refining each interval preserves the composite}{Splitting each piece into $m$ equal pieces, for a positive integer $m$,
leaves its composite unchanged by \itemref{1.4} and functoriality.
Thus $L_{mN}(\gamma)=L_N(\gamma)$.}
\proofstep{4.2-s05-refinement}{Compare equal subdivisions through their common refinement}{Fix a fine subdivision with $N_0$ pieces.  Any other fine subdivision,
with $N$ pieces, has a common refinement with $NN_0$ pieces.  Hence
$L_N(\gamma)=L_{NN_0}(\gamma)=L_{N_0}(\gamma)$.}
Write $L(\gamma)$ for this common value.

Let $H$ be a homotopy from $\gamma$ to $\delta$ fixing the endpoints.
\proofstep{4.2-s06-grid}{Choose a fine grid for the homotopy}{Apply \itemref{1.1} to the compact square $\uI^2$ with its maximum metric
and the neighborhoods $H^{-1}(U_i)$.  Choose an $N\times N$ grid whose mesh
is smaller than the resulting Lebesgue number.  Each closed cell maps
into one $U_i$.}
\proofstep{4.2-s07-cell}{The two routes around one cell have the same value}{In that subspace the two routes between opposite vertices of a cell are
homotopic: map the straight-line homotopy between the two boundary
routes of the cell through $H$.  Applying $F_i$ therefore makes the
cell square commute.}
The values on a shared edge agree even when adjacent cells use different
indices, by compatibility on intersections.
\proofstep{4.2-s08-paste}{Paste the commuting squares across one row}{Pasting the cell squares across one row equates its two boundary composites.}
\proofstep{4.2-s09-boundary}{Cancel the constant endpoint boundary paths}{Since $H$ fixes the endpoints, the outer vertical paths are constant
and have identity values.  Cancel them to equate the top and bottom row values.}
\proofstep{4.2-s10-rows}{Repeat the row equality throughout the homotopy}{Repeat through all rows to obtain $L_N(\gamma)=L_N(\delta)$,
hence $L(\gamma)=L(\delta)$.}

\proofstep{4.2-s11-quotient}{Descend the path value to homotopy classes}{Define $F([\gamma])=L(\gamma)$.  The preceding equality makes this
well defined on homotopy classes.}
A constant path has identity value.  For composable paths, choose the
same fine $N$ for both; concatenation then uses $2N$ pieces and gives
$L(\gamma\ptrans\delta)=L(\delta)\circ L(\gamma)$.
Thus $F$ is a functor.
\proofstep{4.2-s12-restrict}{Collapse the subdivision inside a single cover member}{If the whole path $\gamma$ lies in $U_i$, value every piece using $F_i$;
functoriality collapses the composite to $F_i([\gamma])$.}
This also proves \itemref{4.3}.  Specializing to a cover with two members
gives \itemref{4.4} and \itemref{4.5}.
\end{proofpassage}

\begin{proposition}[id=4.3]{The glued functor restricts to the given functors}
Suppose that \hypCoverFamily, let the $F_{i}$ agree after restriction along the
inclusions of the pairwise intersections, and let $F$ be the functor of
\itemref{4.2} glued from them.  Then $F \circ \fgpd{\incl{U_{i}}} = F_{i}$ for
every $i$.
\leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glue}
\end{proposition}

For the two-set case, let $A$ and $B$ be subsets of $X$.

\begin{definition}[id=4.4]{Gluing two functors out of fundamental groupoids}
Suppose that \hypCoverTwo.  Let $F_{A}$ and $F_{B}$ be functors from
$\fgpd{A}$ and $\fgpd{B}$ to $\mathcal{D}$, and suppose that
\[
  F_{A} \circ \fgpd{\inclin{A \cap B}{A}}
    = F_{B} \circ \fgpd{\inclin{A \cap B}{B}}.
\]
Then $F_{A}$ and $F_{B}$ glue to a functor $F$ from $\fgpd{X}$ to
$\mathcal{D}$.
\leanref{TauCeti.FundamentalGroupoid.glueTwo}
\end{definition}

\begin{proposition}[id=4.5]{The functor glued from two restricts to each of them}
Suppose that \hypCoverTwo, let $F_{A}$ and $F_{B}$ agree after restriction along
the inclusions of $A \cap B$, and let $F$ be the functor of \itemref{4.4} glued
from them.
\itempart{a}{%
  $F \circ \fgpd{\incl{A}} = F_{A}$.
  \leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glueTwo_left}}
\itempart{b}{%
  $F \circ \fgpd{\incl{B}} = F_{B}$.
  \leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glueTwo_right}}
\end{proposition}
