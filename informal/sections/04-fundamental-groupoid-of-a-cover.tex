\begin{theorem}[id=4.1]{Covers generate the fundamental groupoid}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily.  Then the subgroupoids $\im \fgpd{\incl{U_{i}}}$
generate $\fgpd{X}$.
\leanref{TauCeti.FundamentalGroupoid.iSup_im_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{4.1-subdivide}{Proof}{Subdivide and reassemble}
Let $\mathcal H$ be the subgroupoid generated by the images in the
statement, and let $\omega$ be a morphism of $\fgpd X$.  Choose a path
$\gamma$ representing $\omega$.  \proofstep{4.1-s01-subdivide}{Choose a subdivision subordinate to the cover}{Pulling the covering neighborhoods back
along $\gamma$ gives the hypothesis of \itemref{1.7}; hence choose
$0=t_0\le\cdots\le t_n=1$ and indices $i_k$ such that
$\gamma_k=\subpath{\gamma}{t_k}{t_{k+1}}$ lies in $U_{i_k}$.}
\proofstep{4.1-s02-image}{Each piece lies in an image subgroupoid}{By \itemref{1.4}, mapping $\codres{\gamma_k}{U_{i_k}}$ back to $X$ recovers
$\gamma_k$.  Thus $[\gamma_k]$ lies in the corresponding image subgroupoid,
and hence in $\mathcal H$.}  Repeatedly applying \itemref{1.8} gives
\proofequation{4.1-s03-reassemble}{Reassemble the original path class}{  \omega=[\gamma]=[\gamma_0]\ptrans\cdots\ptrans[\gamma_{n-1}].}
The right side lies in $\mathcal H$ by closure under composition.  Every
morphism therefore lies in $\mathcal H$, which proves the claim.
\end{proofpassage}

\begin{definition}[id=4.2]{Gluing functors out of fundamental groupoids}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily.  For each $i$ let $F_{i}$ be a functor from
$\fgpd{U_{i}}$ to $\mathcal{D}$, and suppose that for all $i$ and $j$
\[
  F_{i} \circ \fgpd{\inclin{U_{i} \cap U_{j}}{U_{i}}}
    = F_{j} \circ \fgpd{\inclin{U_{i} \cap U_{j}}{U_{j}}}.
\]
Then the $F_{i}$ glue to a functor $F$ from $\fgpd{X}$ to $\mathcal{D}$.
\leanref{TauCeti.FundamentalGroupoid.glue}
\end{definition}

\begin{proofpassage}{4.2-homotopy}{Construction}{Gluing on homotopy classes}
For a point $y$, choose a cover member containing it and put
$F(y)=F_i(y)$.  \proofstep{4.2-s01-objects}{The object value is independent of the cover member}{This does not depend on $i$: if $y\in U_i\cap U_j$,
the assumed equality of functors on the intersection gives
$F_i(y)=F_j(y)$.}

For a path $\gamma$ from $y$ to $z$, choose a subdivision as in
\itemref{1.7}, with pieces $\gamma_k$ lying in $U_{i_k}$, and set
\[
  L(\gamma)=F_{i_{n-1}}([\gamma_{n-1}])\circ\cdots
             \circ F_{i_0}([\gamma_0]).
\]
The object equalities above make the factors composable.  \proofstep{4.2-s02-indices}{Compatibility makes the local path value independent of the index}{If a piece lies
in both $U_i$ and $U_j$, regard it as a path in their intersection; the
compatibility hypothesis gives the same value under $F_i$ and $F_j$.}
Splitting a piece changes no value, by \itemref{1.8} and functoriality of
$F_i$.  \proofstep{4.2-s03-refinement}{Compare subdivisions through a common refinement}{Two subdivisions have a common refinement, so $L(\gamma)$ is
independent of both the subdivision and its chosen indices.}

Let $H$ be a homotopy from $\gamma$ to $\delta$ fixing the endpoints.
\proofstep{4.2-s04-grid}{Choose a fine grid for the homotopy}{Apply \itemref{1.1} to the compact square $\uI^2$ with its maximum metric
and the neighborhoods $H^{-1}(U_i)$.  Choose a uniform square grid with
mesh smaller than the resulting Lebesgue number.  Each closed cell maps
into one $U_i$.}  \proofstep{4.2-s05-cell}{The two routes around one cell have the same value}{In that subspace the two routes between opposite vertices
of the cell are homotopic: use the two boundary paths of the square and
map their endpoint-preserving homotopy through $H$.  Applying $F_i$
therefore makes the corresponding cell square commute.}  The values on a
shared edge agree even when its adjacent cells use different indices,
by compatibility on intersections.  \proofstep{4.2-s06-paste}{Paste the commuting cell squares}{Composing the cell squares across a
row, and then over all rows, gives equality of the composites along the
two outer boundary routes.}  \proofstep{4.2-s07-boundary}{The endpoint boundary paths contribute identities}{The outer vertical paths are constant because
$H$ fixes its endpoints; their values are identities.}  Consequently
$L(\gamma)=L(\delta)$.

\proofstep{4.2-s08-quotient}{Descend the path value to homotopy classes}{Define $F([\gamma])=L(\gamma)$.  The preceding equality makes this
well defined on homotopy classes.}  A constant path has value an identity.
For composable paths, subdivide each and concatenate the subdivisions;
the defining composite gives
$L(\gamma\ptrans\delta)=L(\delta)\circ L(\gamma)$.
Thus $F$ is a functor.  \proofstep{4.2-s09-restrict}{Use a single cover member to recover the local functor}{For a path lying in $U_i$ use the single-piece
subdivision with index $i$; its value is $F_i([\gamma])$.}  This also proves
\itemref{4.3}.  Taking the two cover members $A$ and $B$ gives
\itemref{4.4} and \itemref{4.5}.
\end{proofpassage}

\begin{proposition}[id=4.3]{The glued functor restricts to the given functors}
Suppose that \hypCoverFamily, let the $F_{i}$ agree after restriction along the
inclusions of the pairwise intersections, and let $F$ be the functor of
\itemref{4.2} glued from them.  Then $F \circ \fgpd{\incl{U_{i}}} = F_{i}$ for
every $i$.
\leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glue}
\end{proposition}

\begin{definition}[id=4.4]{Gluing two functors out of fundamental groupoids}
Suppose that \hypCoverTwo.  Let $F_{A}$ and $F_{B}$ be functors from
$\fgpd{A}$ and $\fgpd{B}$ to $\mathcal{D}$, and suppose that
\[
  F_{A} \circ \fgpd{\inclin{A \cap B}{A}}
    = F_{B} \circ \fgpd{\inclin{A \cap B}{B}}.
\]
Then $F_{A}$ and $F_{B}$ glue to a functor $F$ from $\fgpd{X}$ to
$\mathcal{D}$.
\leanref{TauCeti.FundamentalGroupoid.glueTwo}
\end{definition}

\begin{proposition}[id=4.5]{The functor glued from two restricts to each of them}
Suppose that \hypCoverTwo, let $F_{A}$ and $F_{B}$ agree after restriction along
the inclusions of $A \cap B$, and let $F$ be the functor of \itemref{4.4} glued
from them.
\itempart{a}{%
  $F \circ \fgpd{\incl{A}} = F_{A}$.
  \leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glueTwo_left}}
\itempart{b}{%
  $F \circ \fgpd{\incl{B}} = F_{B}$.
  \leanref{TauCeti.FundamentalGroupoid.map_subtypeVal_comp_glueTwo_right}}
\end{proposition}
