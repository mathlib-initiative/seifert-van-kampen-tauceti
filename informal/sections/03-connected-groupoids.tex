\begin{definition}[id=3.1]{The functor picking out a chosen object}
Let $\mathcal{C}$ be a category, and let $w$ be an object of $\mathcal{C}$.
There is a functor from $\sobj{\End(w)}$ to $\mathcal{C}$ that takes the one
object of $\sobj{\End(w)}$ to $w$, and that takes a morphism $g$ of
$\sobj{\End(w)}$, which is an element of $\End(w)$, to $g$ itself.
\leanref{TauCeti.Groupoid.singleObjFunctor}
\end{definition}

\begin{definition}[id=3.2]{The functor induced by a homomorphism out of a vertex group}
Let $\mathcal{C}$ be a groupoid, let $w$ be an object of $\mathcal{C}$, and
let $\tau$ give, for each object $c$ of $\mathcal{C}$, a morphism $\tau_{c}$
from $w$ to $c$.  Let $M$ be a monoid, and let $\varphi$ be a homomorphism of
monoids from $\End(w)$ to $M$.  There is a functor $F$ from $\mathcal{C}$ to
$\sobj{M}$ that takes every object of $\mathcal{C}$ to the one object of
$\sobj{M}$.  For objects $c$ and $c'$ of $\mathcal{C}$ and a morphism $g$
from $c$ to $c'$, the value of $F$ at $g$ is the image under $\varphi$ of the
element of $\End(w)$ that runs first along $\tau_{c}$, then along $g$, then
along the inverse of $\tau_{c'}$.
\leanref{TauCeti.Groupoid.functorOfEndHom}
\leanref{TauCeti.Groupoid.functorOfEndHom_map}
\end{definition}
