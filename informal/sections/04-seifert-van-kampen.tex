Let $\mathcal C$ be a \fcnotion{CategoryTheory.Category}{category} and $w$ an
object of it.  Write \fcnotion{CategoryTheory.End}{$\End(w)$} for its
endomorphism monoid, and \fcnotion{CategoryTheory.SingleObj}{$\sobj{M}$}
for the one-object category associated to a \fcnotion{Monoid}{monoid} $M$.

\begin{definition}[id=4.1]{The endomorphism functor}
The functor $J_w:\sobj{\End(w)}\longrightarrow\mathcal C$ takes the unique
object to $w$ and each element of $\End(w)$ to itself.
\leanref{TauCeti.Groupoid.singleObjFunctor}
\end{definition}

\begin{definition}[id=4.2]{Extending a vertex-group homomorphism}
Suppose that $\mathcal C$ is a groupoid and that a morphism
$\tau_c:w\longrightarrow c$ has been chosen for each object $c$.  Let $M$ be
a monoid and $f:\End(w)\longrightarrow M$ a homomorphism.
The induced functor $F_{\tau,f}:\mathcal C\longrightarrow\sobj{M}$ acts on a
morphism $g:c\longrightarrow c'$ by
\begin{equation}\label{eq:4.2-vertex-group-functor}
  F_{\tau,f}(g)=f\bigl(\tau_{c'}^{-1}\circ g\circ\tau_c\bigr).
\end{equation}
\leanref{TauCeti.Groupoid.functorOfEndHom}
\leanref{TauCeti.Groupoid.functorOfEndHom_map}
\end{definition}

\begin{samepage}
\begin{theorem}[id=4.3]{The descent homomorphism}
Let $A,B\subseteq X$ for a topological space $X$, let \hypBasepoint, and let
$M$ be a monoid.  Suppose that \hypCoverTwo and \hypPathConnected.
Let $\varphi_A$ and $\varphi_B$ be homomorphisms such that the square
\[
  \begin{array}{ccc@{}l}
    \fgrp{A\cap B}{x} & \longrightarrow & \fgrp{A}{x} & \\
    \big\downarrow && \big\downarrow & \;\scriptstyle\varphi_A \\
    \fgrp{B}{x} & \xrightarrow{\ \varphi_B\ } & M &
  \end{array}
\]
commutes, with the unlabelled maps induced by inclusions.  Then there is a unique
homomorphism $\varphi:\fgrp{X}{x}\longrightarrow M$ such that
\[
  \varphi\circ(\incl A)_*=\varphi_A,\qquad
  \varphi\circ(\incl B)_*=\varphi_B.
\]
\leanref{TauCeti.vanKampenDesc}
\leanref{TauCeti.vanKampenDesc_comp_map_left}
\leanref{TauCeti.vanKampenDesc_comp_map_right}
\leanref{TauCeti.vanKampen_hom_ext}
\end{theorem}
\end{samepage}

\begin{proofpassage}{4.3-descent}{Proof}{Existence and uniqueness}
\proofstep{4.3-s01-connect}{Choose connecting paths compatibly on the intersection}{By path connectedness, for each $y\in X$ choose a path $\tau_y^A:x\longrightarrow y$
with image in $A$ if $y\in A$, and $\tau_y^B$ with image in $B$ if $y\in B$.
For $y\in A\cap B$, we may choose $\tau_y^A=\tau_y^B$ lying in $A\cap B$,
taking the constant path when $y=x$.}
\proofstep{4.3-s02-local}{Invoke the functor construction from a homomorphism}{Using the classes of these paths in \itemref{4.2}, the homomorphisms
$\varphi_A,\varphi_B$ give functors $F_A:\fgpd A\longrightarrow\sobj M$ and
$F_B:\fgpd B\longrightarrow\sobj M$.}

For a path $\gamma:y\longrightarrow z$ in $A\cap B$, we have
$\tau_y^A=\tau_y^B$ and $\tau_z^A=\tau_z^B$. Thus
\eqref{eq:4.2-vertex-group-functor} applies $\varphi_A$ and $\varphi_B$ to the
class of the same based loop in $A\cap B$. By compatibility,
$F_A([\gamma])=F_B([\gamma])$, so the square commutes.
\proofequation{4.3-s03-compatibility}{The local functors agree on the intersection}{
  \begin{array}{ccc@{}l}
    \fgpd{A\cap B} & \longrightarrow & \fgpd A & \\
    \big\downarrow && \big\downarrow & \;\scriptstyle F_A \\
    \fgpd B & \xrightarrow{\ F_B\ } & \sobj M &
  \end{array}.
}
\proofstep{4.3-s04-glue}{Invoke gluing of the two local functors}{Thus \itemref{2.2} glues $F_A,F_B$ to a functor
$F:\fgpd X\longrightarrow\sobj M$.}
Restricting to loops at $x$ gives
\proofequation{4.3-s05-endomorphisms}{Restrict the glued functor to loops at the basepoint}{  \varphi:\fgrp{X}{x}\longrightarrow M,\qquad
  \varphi(\omega)=F(\omega).}

Let $\omega\in\fgrp{A}{x}$ be arbitrary.
By the definition of $\varphi$ and the restriction property in
\itemref{2.2},
\proofequation{4.3-s06-left}{Use the glued functor's restriction to A}{
 \varphi\bigl((\incl{A})_*(\omega)\bigr)=F(\fgpd{\incl{A}}(\omega))=F_A(\omega).
}
The constant path $\tau^A_x$ has identity class, so
\proofequation{4.3-s07-basepoint}{Evaluate the local functor at the basepoint}{
 F_A(\omega)=\varphi_A\bigl([\tau^A_x]^{-1}\circ\omega\circ[\tau^A_x]\bigr)
            =\varphi_A(\omega).
}
\proofstep{4.3-s08-pointwise}{Pass from pointwise equality to equality of homomorphisms}{Since this holds for every $\omega$, we have $\varphi\circ(\incl A)_*=\varphi_A$.}
\proofstep{4.3-s09-right}{Apply the right restriction in the same way}{Replacing $A$ by $B$ gives $\varphi\circ(\incl B)_*=\varphi_B$.}

For uniqueness, suppose that $\psi:\fgrp{X}{x}\longrightarrow M$ also satisfies
these restriction equations.  Consider
\[
 E=\{\omega\in\fgrp{X}{x}:\varphi(\omega)=\psi(\omega)\}.
\]
It contains the identity and is closed under multiplication, since
$\varphi$ and $\psi$ are homomorphisms.  It is also closed under inverses:
if their values on $\omega$ agree, then $\varphi(\omega^{-1})$ and
$\psi(\omega^{-1})$ are two-sided inverses of the same element of $M$,
and a two-sided inverse is unique.  Thus $E$ is a subgroup, even
though $M$ is only assumed to be a monoid.

\proofstep{4.3-s10-images}{Both generating images lie in the equalizer subgroup}{The restriction equations put the images of both $\fgrp{A}{x}$ and $\fgrp{B}{x}$ in $E$.}
\proofstep{4.3-s11-generation}{Invoke generation and conclude that the homomorphisms agree}{These two images generate $\fgrp{X}{x}$ by
\itemref{3.1} applied to the cover $\{A,B\}$.  Therefore $E=\fgrp{X}{x}$: the homomorphisms agree on every loop class,
so $\varphi=\psi$.}
\end{proofpassage}

\begin{theorem}[id=4.4]{The based Seifert--van Kampen theorem}
\fcref{Theorem 1.20}
Let \hypBasepoint, and suppose that \hypCoverTwo and \hypPathConnected.  Then
the square
\[
  \begin{array}{r@{}c@{\quad}c@{\quad}c@{}l}
    & \fgrp{A\cap B}{x} & \xrightarrow{\ j_A\ } & \fgrp{A}{x} & \\
    \scriptstyle j_B\; & \big\downarrow && \big\downarrow & \;\scriptstyle p_A \\
    & \fgrp{B}{x} & \xrightarrow{\ p_B\ } & \fgrp{X}{x} &
  \end{array}
\]
whose homomorphisms are induced by the inclusions is a
\fcnotion{CategoryTheory.IsPushout}{pushout} in the
\fcnotion{GrpCat}{category of groups}.
\leanref{TauCeti.isPushout_fundamentalGroup}
\end{theorem}

\begin{proofpassage}{4.4-pushout}{Proof}{The pushout universal property}
Let $K$ be any group, and let $u_A:\fgrp{A}{x}\longrightarrow K$ and
$u_B:\fgrp{B}{x}\longrightarrow K$ satisfy
$u_A\circ j_A=u_B\circ j_B$.  \proofstep{4.4-s01-existence}{Construct a map for any compatible pair}{By \itemref{4.3}, with
$M=K$, there is a homomorphism $u:\fgrp{X}{x}\longrightarrow K$}
satisfying
\proofequation{4.4-s02-factorization}{Invoke the two restriction equations}{  u\circ p_A=u_A,\qquad u\circ p_B=u_B.}
\proofstep{4.4-s03-uniqueness}{Invoke uniqueness for the mediating homomorphism}{It is unique by \itemref{4.3}.}
This is the universal property of the pushout.
\end{proofpassage}
