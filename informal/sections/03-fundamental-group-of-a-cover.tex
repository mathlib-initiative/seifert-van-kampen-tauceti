For a topological space $X$ with basepoint $x$, the
\fcnotion{FundamentalGroup}{fundamental group} $\fgrp{X}{x}$ is the group
of morphisms from $x$ to itself in $\fgpd{X}$.  Elements of this group are called
\fcnotion{FundamentalGroup.fromPath}{loop classes}.  A continuous map $f:X\longrightarrow Y$ induces a
\fcnotion{MonoidHom}{homomorphism}
\fcnotion[Induced homomorphism $f_*$]{FundamentalGroup.map}{$f_*:\fgrp{X}{x}\longrightarrow\fgrp{Y}{f(x)}$}.

\begin{theorem}[id=3.1]{Generation by subspace groups}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily, that $x$ lies in every $U_i$, and that
$U_{i} \cap U_{j}$ is \fcnotion[Nonempty and path connected]{IsPathConnected}{nonempty and path connected} for all $i$ and $j$.
Then $\fgrp{X}{x}$ is generated by the classes of loops based at $x$
and contained in some $U_i$.
\leanref{TauCeti.FundamentalGroup.iSup_range_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{3.1-basepoint}{Proof}{Generation by based loops}
Let $H$ be the \fcnotion{Subgroup}{subgroup} generated by these classes.
\proofstep{3.1-s01-paths}{Choose paths from the basepoint}{For each point $c$, choose a cover member $U_c$ containing $c$ and, by
path connectedness, a path class $\tau_c:x\longrightarrow c$ inside $U_c$.}
Take $\tau_x$ to be the constant loop class.
For a morphism $\omega:c\longrightarrow d$ of $\fgpd X$, form the loop class
\[
  T(\omega)=\tau_c\ptrans\omega\ptrans\tau_d^{-1}\in\fgrp{X}{x}.
\]
This defines a functor $T:\fgpd X\longrightarrow\fgrp{X}{x}$, viewing the target
as a one-object groupoid: $T(\mathrm{id}_c)=1$, and for composable $\omega,\nu$,
cancellation gives
\[
  T(\nu\circ\omega)=T(\nu)\circ T(\omega).
\]
\proofstep{3.1-s02-subgroupoid}{The transported classes define a subgroupoid}{Hence $S=T^{-1}(H)$ is a subgroupoid of $\fgpd X$.}

Suppose that $\omega$ is represented by a path in $U_i$ from $c$ to $d$.
\proofstep{3.1-s03-intersections}{Choose the two paths in pairwise intersections}{Choose path classes $\sigma_c:x\longrightarrow c$ in
$U_c\cap U_i$ and $\sigma_d:x\longrightarrow d$ in
$U_i\cap U_d$.}  Path connectedness of the intersections supplies these choices.
Cancellation of inverse path classes gives
\proofequation{3.1-s04-factorization}{Factor the transported loop into three local loops}{
 T(\omega)=(\tau_c\ptrans\sigma_c^{-1})
        \ptrans(\sigma_c\ptrans\omega\ptrans\sigma_d^{-1})
        \ptrans(\sigma_d\ptrans\tau_d^{-1}).
}
\proofstep{3.1-s05-middle}{The middle loop lies in the generated subgroup}{The middle loop lies in $U_i$, so its class belongs to $H$.}
\proofstep{3.1-s06-product}{Multiply the three local loop classes}{The first and last factors are loops in $U_c$ and $U_d$,
respectively, so their classes also lie in $H$.  Closure under
multiplication now gives $T(\omega)\in H$.}  \proofstep{3.1-s07-generation}{Invoke generation of the fundamental groupoid}{The subgroupoid $S$ contains all path classes represented in the $U_i$; \itemref{2.1} then gives $S=\fgpd X$.}

\proofstep{3.1-s08-loops}{Every loop class belongs to the subgroup}{Since $\tau_x=1$, every loop class $\omega$ at $x$ satisfies $\omega=T(\omega)\in H$.}
Thus $H=\fgrp{X}{x}$, as required.
\end{proofpassage}
