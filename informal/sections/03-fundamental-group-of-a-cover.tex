For a topological space $X$ with basepoint $x$, the
\fcnotion{FundamentalGroup}{fundamental group} $\fgrp{X}{x}$ is the group
of morphisms from $x$ to itself in $\fgpd{X}$.  A
\fcnotion{FundamentalGroup.fromPath}{loop class at $x$} is an element of
this group.  A continuous map $f:X\longrightarrow Y$ induces a
\fcnotion{MonoidHom}{homomorphism}
\fcnotion[Induced homomorphism $f_*$]{FundamentalGroup.map}{$f_*:\fgrp{X}{x}\longrightarrow\fgrp{Y}{f(x)}$}.
Throughout,
\fcnotion{IsPathConnected}{path-connected subsets} are understood to be
nonempty.

\begin{fcassumptions}
Assume that \hypCoverFamily.

\begin{theorem}[id=3.1]{Generation by subspace groups}
\fcref{Theorem 1.20}
Suppose that $x$ lies in every $U_i$ and that
$U_{i} \cap U_{j}$ is path connected for all $i$ and $j$.  Then the
\fcnotion{Subgroup}{subgroups}
\fcnotion[Image subgroup]{MonoidHom.range}{$\im(\incl{U_i})_*$} generate $\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.iSup_range_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{3.1-basepoint}{Proof}{Generation by based loops}
Let $H$ be the subgroup generated by loop classes from the $U_i$.
\proofstep{3.1-s01-paths}{Choose paths from the basepoint}{For each point $c$, choose an index $i_0(c)$ with $c\in U_{i_0(c)}$ and a
path class $\tau_c:x\longrightarrow c$ inside $U_{i_0(c)}$.}  Each $U_i$ is path connected by hypothesis.
For a morphism $\omega:c\longrightarrow d$ of $\fgpd X$, form the loop class
\[
  T(\omega)=\tau_c\ptrans\omega\ptrans\tau_d^{-1},
\]
\proofstep{3.1-s02-subgroupoid}{The transported classes define a subgroupoid}{The morphisms for which $T(\omega)\in H$ form a
subgroupoid $S$: the identities give the trivial loop, inverses
give $T(\omega)^{-1}$, and the middle factors $\tau_d^{-1}\ptrans\tau_d$
cancel when two such morphisms are composed.}

Suppose that $\omega$ is represented by a path in $U_i$ from $c$ to $d$.
\proofstep{3.1-s03-intersections}{Choose the two paths in pairwise intersections}{Choose path classes $\sigma_c:x\longrightarrow c$ in
$U_{i_0(c)}\cap U_i$ and $\sigma_d:x\longrightarrow d$ in
$U_i\cap U_{i_0(d)}$.}  Path connectedness of the intersections supplies these choices.
Cancellation of inverse path classes gives
\proofequation{3.1-s04-factorization}{Factor the transported loop into three local loops}{\begin{aligned}
 T(\omega)
   &=(\tau_c\ptrans\sigma_c^{-1})
        \ptrans(\sigma_c\ptrans\omega\ptrans\sigma_d^{-1})\\
   &\qquad\ptrans(\sigma_d\ptrans\tau_d^{-1}).
\end{aligned}}
\proofstep{3.1-s05-middle}{The middle loop lies in the generated subgroup}{The middle loop lies in $U_i$, so its class belongs to $H$.}
\proofstep{3.1-s06-product}{Multiply the three local loop classes}{The first and last factors are loops in $U_{i_0(c)}$ and $U_{i_0(d)}$,
respectively, so their classes also lie in $H$.  Closure under
multiplication now gives $T(\omega)\in H$.}  \proofstep{3.1-s07-generation}{Invoke generation of the fundamental groupoid}{The subgroupoid $S$ contains all path classes represented in the $U_i$; \itemref{2.1} implies $S=\top$, the whole fundamental groupoid.}

For a loop class $\omega$ at $x$ we now have $T(\omega)\in H$.
\proofstep{3.1-s08-basepoint-loop}{The chosen path at the basepoint belongs to the subgroup}{The class $\tau_x$ is itself a loop inside $U_{i_0(x)}$, so it lies in $H$.}
Hence
\proofequation{3.1-s09-cancellation}{Cancel the basepoint transport}{  \omega=\tau_x^{-1}\ptrans T(\omega)\ptrans\tau_x\in H.}
Thus $H=\fgrp{X}{x}$, as required.
\end{proofpassage}

\end{fcassumptions}

\begin{corollary}[id=3.2]{Generation from two subspaces}
Let $A,B\subseteq X$.  Suppose that \hypCoverTwo, that \hypPathConnected,
and that \hypBasepoint.
Then $\im (\incl{A})_*$ and $\im (\incl{B})_*$ generate
$\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.range_map_subtypeVal_sup_eq_top}
\end{corollary}
