Let $X$ be a topological space, $x\in X$, and $(U_i)_{i\in I}$ a family of
subsets of $X$.
The \fcnotion{FundamentalGroup}{fundamental group}
$\fgrp{X}{x}$ is the group of morphisms from $x$ to itself in $\fgpd{X}$.
Multiplication is composition, with the right factor traversed first.
The \fcnotion{FundamentalGroup.fromPath}{class of a loop at $x$} is an element
of this group.  A continuous map $f:X\longrightarrow Y$ induces a
\fcnotion{MonoidHom}{homomorphism}
\fcnotion{FundamentalGroup.map}{$\fgrp{f}{x}$} from $\fgrp{X}{x}$ to
$\fgrp{Y}{f(x)}$.

Write $\incl{S}$ for a subspace inclusion.  The
\fcnotion{MonoidHom.range}{image} of a group homomorphism is the
\fcnotion{Subgroup}{subgroup} of its values; subgroups generate when the
smallest subgroup containing them is the whole group.  Here
\fcnotion{IsPathConnected}{path-connected subsets} are understood to be
nonempty.

\begin{theorem}[id=4.1]{Covers generate the fundamental group}
\fcref{Theorem 1.20}
Suppose that \hypCoverFamily, that $x$ lies in every $U_{i}$, and that
$U_{i} \cap U_{j}$ is path connected for all $i$ and $j$.  Then the subgroups
$\im \fgrp{\incl{U_{i}}}{x}$ generate $\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.iSup_range_map_subtypeVal_eq_top}
\end{theorem}

\begin{proofpassage}{4.1-basepoint}{Proof}{Generation by based loops}
Let $H$ be the subgroup generated by the images in the statement.
\proofstep{4.1-s01-paths}{Choose paths from the basepoint}{For each point $c$, choose an index $i_0(c)$ with $c\in U_{i_0(c)}$ and a
path class $\tau_c:x\longrightarrow c$ inside $U_{i_0(c)}$.}  Such a path
exists since $x\in U_{i_0(c)}$ and $U_{i_0(c)}=U_{i_0(c)}\cap U_{i_0(c)}$ is
path connected.  For a morphism $\omega:c\longrightarrow d$ of
$\fgpd X$ write
\[
  T(\omega)=\tau_c\ptrans\omega\ptrans\tau_d^{-1},
\]
a loop class at $x$.  \proofstep{4.1-s02-subgroupoid}{The transported classes define a subgroupoid}{The morphisms for which $T(\omega)\in H$ form a
subgroupoid $S$: the identities give the trivial loop, inverses
give $T(\omega)^{-1}$, and the middle factors $\tau_d^{-1}\ptrans\tau_d$
cancel when two such morphisms are composed.}

Suppose that $\omega$ is represented by a path in $U_i$ from $c$ to $d$.
\proofstep{4.1-s03-intersections}{Choose the two paths in pairwise intersections}{Choose path classes $\sigma_c:x\longrightarrow c$ in
$U_{i_0(c)}\cap U_i$ and $\sigma_d:x\longrightarrow d$ in
$U_i\cap U_{i_0(d)}$.}  Both intersections contain $x$ and the required
endpoint and are path connected.  Cancellation of inverse path classes
gives
\proofequation{4.1-s04-factorization}{Factor the transported loop into three local loops}{\begin{aligned}
 T(\omega)
   &=(\tau_c\ptrans\sigma_c^{-1})
        \ptrans(\sigma_c\ptrans\omega\ptrans\sigma_d^{-1})\\
   &\qquad\ptrans(\sigma_d\ptrans\tau_d^{-1}).
\end{aligned}}
\proofstep{4.1-s05-middle}{The middle loop lies in the generated subgroup}{The middle factor is represented by a loop wholly in $U_i$, since all
three of its path segments lie there.  Its class therefore lies in $H$.}
\proofstep{4.1-s06-product}{Multiply the three local loop classes}{The first and last factors are loops in $U_{i_0(c)}$ and $U_{i_0(d)}$,
respectively, so their classes also lie in $H$.  Closure under
multiplication now gives $T(\omega)\in H$.}  \proofstep{4.1-s07-generation}{Invoke generation of the fundamental groupoid}{The subgroupoid $S$ contains every image
of a cover member; \itemref{3.1} implies $S=\top$, the whole fundamental groupoid.}

For a loop class $\omega$ at $x$ we now have $T(\omega)\in H$.
\proofstep{4.1-s08-basepoint-loop}{The chosen path at the basepoint belongs to the subgroup}{The class $\tau_x$ is itself a loop inside $U_{i_0(x)}$, so it lies in $H$.}
Hence
\proofequation{4.1-s09-cancellation}{Cancel the basepoint transport}{  \omega=\tau_x^{-1}\ptrans T(\omega)\ptrans\tau_x\in H.}
Thus $H=\fgrp{X}{x}$, as required.  For a two-member family, the pairwise
intersections are its two members and their intersection in either order.
The hypotheses of \itemref{4.2} therefore give those just used, proving that
corollary as well.
\end{proofpassage}

For the two-set case, let $A$ and $B$ be subsets of $X$.

\begin{corollary}[id=4.2]{Two sets generate the fundamental group}
Suppose that \hypCoverTwo, that \hypPathConnected, and that \hypBasepoint.
Then $\im \fgrp{\incl{A}}{x}$ and $\im \fgrp{\incl{B}}{x}$ generate
$\fgrp{X}{x}$.
\leanref{TauCeti.FundamentalGroup.range_map_subtypeVal_sup_eq_top}
\end{corollary}
